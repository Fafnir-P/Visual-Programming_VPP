import os, sys
d = os.path.dirname(os.path.abspath(__file__))
while not os.path.isdir(os.path.join(d, 'card_templates')) and os.path.dirname(d) != d: d = os.path.dirname(d)
sys.path.insert(0, d)
import argparse
import json

from card_templates.runtime import Context, Store, CardError
from cards import (enter_text, trim_whitespace, change_text_case, check_field_is_required,
                   enter_number, calculate_from_two_values, format_currency_value)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--customer', default=None)
    ap.add_argument('--qty', default=None)
    ap.add_argument('--price', default=None)
    args = ap.parse_args()
    ctx = Context(store=Store())
    try:
        n2 = enter_text(ctx, value=args.customer, name='customer', label='Customer')
        n3 = trim_whitespace(ctx, text=n2['value'])
        n4 = change_text_case(ctx, text=n3['text'], case='title')
        n5 = check_field_is_required(ctx, value=n4['text'], field='customer')
        n6 = enter_number(ctx, value=args.qty, name='qty', label='Qty')
        n7 = enter_number(ctx, value=args.price, name='price', label='Price')
        n8 = calculate_from_two_values(ctx, first=n6['value'], second=n7['value'], operation='multiply', places=2)
        n9 = format_currency_value(ctx, value=n8['result'], symbol='$', separator=',', decimals=2)
    except CardError as e:
        print(e, file=sys.stderr)
        sys.exit(1)
    print(json.dumps({'customer': n5['value'], 'total': n9['text']}))


if __name__ == '__main__':
    main()
